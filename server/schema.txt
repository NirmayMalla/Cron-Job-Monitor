SCHEMAS FOR CRONJOB:

Montior:
create table monitors(id SERIAL PRIMARY KEY, name VARCHAR(100) NOT NULL, expected_interval INT NOT NULL, grace_period INT NOT NULL, status VARCHAR(20) NOT NULL, monitor_key VARCHAR(100) NOT NULL);

Ping:
create table pings(id SERIAL PRIMARY KEY, monitor_id INTEGER REFERENCES monitors(id) NOT NULL, state VARCHAR(30) NOT NULL, timestamp TIMESTAMP NOT NULL, duration INT NOT NULL);
